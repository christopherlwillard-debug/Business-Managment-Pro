    <script>
        // Tailwind script already loaded via CDN
        function initTailwind() {

        // Set default baseboard rate based on type
        function setBaseRate() {
            const type = document.getElementById('base_type').value;
            const rateInput = document.getElementById('base_rate');
            if (type === 'basic') rateInput.value = 8;
            else if (type === 'standard') rateInput.value = 12.5;
            else if (type === 'large') rateInput.value = 20;
            document.getElementById('base_chosen').innerHTML = 
                `${document.getElementById('base_type').selectedOptions[0].text.split('→')[0].trim()} • $${rateInput.value}/ft chosen`;
            calculate();
        }

        // Set default crown rate based on type
        function setCrownRate() {
            const type = document.getElementById('crown_type').value;
            const rateInput = document.getElementById('crown_rate');
            if (type === 'basic') rateInput.value = 12.5;
            else if (type === 'standard') rateInput.value = 18.5;
            else if (type === 'large') rateInput.value = 28.5;
            document.getElementById('crown_chosen').innerHTML = 
                `${document.getElementById('crown_type').selectedOptions[0].text.split('→')[0].trim()} • $${rateInput.value}/ft chosen`;
            calculate();
        }

        // Set finish rate from buttons
        function setFinish(rate) {
            document.getElementById('finish_rate').value = rate;
            // Highlight active button
            document.querySelectorAll('.finish-btn').forEach(btn => {
                btn.dataset.active = (parseFloat(btn.getAttribute('onclick').match(/\d+\.?\d*/)[0]) === rate);
            });
            calculate();
        }

        // Main calculation engine
        function calculate() {
            // Get inputs
            const length = parseFloat(document.getElementById('length').value) || 0;
            const width = parseFloat(document.getElementById('width').value) || 0;
            const linearFeet = 2 * (length + width);

            // Display linear feet
            document.getElementById('perimeter-display').textContent = `${linearFeet} ft`;
            document.getElementById('linear-feet-display').textContent = `${linearFeet} ft`;

            // Rates
            const baseRate = parseFloat(document.getElementById('base_rate').value) || 0;
            const crownRate = parseFloat(document.getElementById('crown_rate').value) || 0;
            const finishRate = parseFloat(document.getElementById('finish_rate').value) || 0;

            // Base calculations
            let baseLabor = linearFeet * baseRate;
            let crownLabor = linearFeet * crownRate;
            let finishLabor = linearFeet * finishRate;
            let trimLabor = baseLabor + crownLabor + finishLabor;

            // Vaulted ceiling multiplier
            const vaulted = document.getElementById('vaulted').checked;
            const vaultPct = parseFloat(document.getElementById('vault_pct').value) || 30;
            if (vaulted) {
                trimLabor = trimLabor * (1 + vaultPct / 100);
            }

            // Add-ons
            const outsideNum = parseFloat(document.getElementById('outside_num').value) || 0;
            const outsidePrice = parseFloat(document.getElementById('outside_price').value) || 20;
            const non90Num = parseFloat(document.getElementById('non90_num').value) || 0;
            const non90Price = parseFloat(document.getElementById('non90_price').value) || 37.5;
            const repairs = parseFloat(document.getElementById('repairs').value) || 0;
            const demo = parseFloat(document.getElementById('demo').value) || 0;

            const addonsTotal = (outsideNum * outsidePrice) + (non90Num * non90Price) + repairs + demo;

            // Materials
            const materials = parseFloat(document.getElementById('materials').value) || 0;

            // Subtotal
            const subtotal = trimLabor + addonsTotal + materials;

            // Final price logic
            const minCharge = 500; // Pro note default
            let roundedSubtotal = Math.ceil(subtotal / 50) * 50;
            const finalPrice = Math.max(roundedSubtotal, minCharge);
            const adjustment = finalPrice - subtotal;

            // Update all displays
            document.getElementById('base-labor-display').textContent = `$${(baseLabor).toFixed(0)}`;
            document.getElementById('crown-labor-display').textContent = `$${(crownLabor).toFixed(0)}`;
            document.getElementById('finish-labor-display').textContent = `$${(finishLabor).toFixed(0)}`;

            document.getElementById('base-labor-final').textContent = `$${(baseLabor).toFixed(0)}`;
            document.getElementById('crown-labor-final').textContent = `$${(crownLabor).toFixed(0)}`;
            document.getElementById('finish-labor-final').textContent = `$${(finishLabor).toFixed(0)}`;
            document.getElementById('materials-final').textContent = `$${(materials).toFixed(0)}`;
            document.getElementById('addons-final').textContent = `$${(addonsTotal).toFixed(0)}`;

            document.getElementById('subtotal-display').textContent = `$${(subtotal).toFixed(0)}`;
            document.getElementById('adjustment-display').textContent = adjustment > 0 ? `+$${(adjustment).toFixed(0)}` : '$0';
            document.getElementById('final-price-display').innerHTML = 
                `$${(finalPrice).toFixed(0)}<span class="text-base font-normal text-zinc-400 ml-2">incl. round-up</span>`;

            document.getElementById('addons-total-display').textContent = `$${(addonsTotal).toFixed(0)}`;

            // Quick version
            const quickTotal = linearFeet * (baseRate + crownRate + finishRate) + materials + addonsTotal;
            document.getElementById('quick-version-display').innerHTML = 
                `${linearFeet} ft × ($${baseRate} + $${crownRate} + $${finishRate}) + $${materials} + $${addonsTotal} = <span class="font-bold">$${(quickTotal).toFixed(0)}</span>`;
        }

        // Reset a single section
        function resetSection(section) {
            if (section === 1) {
                document.getElementById('length').value = 20;
                document.getElementById('width').value = 15;
                document.getElementById('ceiling').value = 9;
            }
            calculate();
        }

        // Copy full estimate to clipboard
        function copyEstimate() {
            const length = document.getElementById('length').value;
            const width = document.getElementById('width').value;
            const lf = (2 * (parseFloat(length) + parseFloat(width))).toFixed(0);
            const final = document.getElementById('final-price-display').textContent.replace(/[^0-9]/g, '');
            
            const text = `🧰 Trim Estimate\n` +
                        `Room: ${length}×${width} ft (${lf} linear ft)\n` +
                        `Baseboard: ${document.getElementById('base_chosen').innerText}\n` +
